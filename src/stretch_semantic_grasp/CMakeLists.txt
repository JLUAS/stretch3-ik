cmake_minimum_required(VERSION 3.8)
project(stretch_semantic_grasp)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(semantic_grasp)
install(PROGRAMS scripts/semantic_grasp_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(semantic_grasp_core tests WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    APPEND_ENV "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 60)
endif()
ament_package()
