cmake_minimum_required(VERSION 3.16)
project(stretch_guarded_executor)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(stretch_custom_kinematics REQUIRED)
find_package(stretch_robot_interface REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(action_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_srvs REQUIRED)
find_package(ament_index_cpp REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "action/ExecuteLiftTarget.action")
add_library(guard SHARED src/guard.cpp src/diagnostic_policy.cpp src/lift_sequence.cpp)
target_include_directories(guard PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(guard stretch_custom_kinematics trajectory_msgs diagnostic_msgs action_msgs)
add_executable(guarded_executor src/executor_node.cpp)
target_link_libraries(guarded_executor guard)
ament_target_dependencies(guarded_executor stretch_robot_interface rclcpp_action control_msgs action_msgs diagnostic_msgs rcl_interfaces std_srvs ament_index_cpp)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(guarded_executor "${cpp_typesupport_target}")
install(TARGETS guard EXPORT export_guard ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS guarded_executor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_guard test/test_guard.cpp)
  target_link_libraries(test_guard guard)
  ament_add_gtest(test_lift_sequence test/test_lift_sequence.cpp)
  target_link_libraries(test_lift_sequence guard)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  ament_add_test(test_dry_transport
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/test_guarded_executor_dry.py
      --output ${CMAKE_CURRENT_BINARY_DIR}/dry_transport.json
    ENV ROS_DOMAIN_ID=20 ROS_LOCALHOST_ONLY=1
    TIMEOUT 60 GENERATE_RESULT_FOR_RETURN_CODE_ZERO)
  ament_add_test(test_lift_target_transport
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/test_lift_target_transport.py
    ENV ROS_DOMAIN_ID=20 ROS_LOCALHOST_ONLY=1
    TIMEOUT 140 GENERATE_RESULT_FOR_RETURN_CODE_ZERO)
endif()
ament_export_targets(export_guard HAS_LIBRARY_TARGET)
ament_export_dependencies(stretch_custom_kinematics trajectory_msgs diagnostic_msgs action_msgs)
ament_export_dependencies(rosidl_default_runtime)
ament_package()
