cmake_minimum_required(VERSION 3.16)
project(stretch_custom_kinematics)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(urdf REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(stretch_robot_interface REQUIRED)
find_package(interactive_markers REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)

add_library(${PROJECT_NAME} SHARED src/forward_kinematics.cpp src/inverse_kinematics.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(${PROJECT_NAME} Eigen3::Eigen)
ament_target_dependencies(${PROJECT_NAME} rclcpp urdf sensor_msgs geometry_msgs tf2 tf2_ros stretch_robot_interface)
add_executable(fk_validator src/fk_validator.cpp)
target_link_libraries(fk_validator ${PROJECT_NAME})
ament_target_dependencies(fk_validator rclcpp urdf sensor_msgs geometry_msgs tf2 tf2_ros stretch_robot_interface)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
add_library(ik_status_panel SHARED src/ik_status_panel.cpp include/stretch_custom_kinematics/ik_status_panel.hpp)
set_target_properties(ik_status_panel PROPERTIES AUTOMOC ON)
target_include_directories(ik_status_panel PRIVATE include)
target_link_libraries(ik_status_panel Qt5::Widgets)
ament_target_dependencies(ik_status_panel rviz_common pluginlib rclcpp diagnostic_msgs geometry_msgs std_srvs)
pluginlib_export_plugin_description_file(rviz_common rviz_plugins.xml)
install(TARGETS ik_status_panel LIBRARY DESTINATION lib)
add_executable(ik_preview src/ik_preview.cpp)
target_link_libraries(ik_preview ${PROJECT_NAME})
ament_target_dependencies(ik_preview interactive_markers visualization_msgs diagnostic_msgs ament_index_cpp std_srvs)
install(TARGETS ik_preview DESTINATION lib/${PROJECT_NAME})
add_executable(ik_benchmark src/ik_benchmark.cpp)
target_link_libraries(ik_benchmark ${PROJECT_NAME})
install(TARGETS ik_benchmark DESTINATION lib/${PROJECT_NAME})
install(TARGETS fk_validator DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS src/offline_seed.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch rviz DESTINATION share/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_ik test/test_ik.cpp)
  target_link_libraries(test_ik ${PROJECT_NAME})
  target_compile_definitions(test_ik PRIVATE TEST_URDF="${CMAKE_CURRENT_SOURCE_DIR}/test/robot.urdf" TEST_LIMITS="${CMAKE_CURRENT_SOURCE_DIR}/config/hardware_limits.csv")
  ament_add_gtest(test_${PROJECT_NAME} test/test_core.cpp)
  target_link_libraries(test_${PROJECT_NAME} ${PROJECT_NAME})
  target_compile_definitions(test_${PROJECT_NAME} PRIVATE TEST_URDF="${CMAKE_CURRENT_SOURCE_DIR}/test/robot.urdf")
endif()
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp Eigen3 urdf sensor_msgs geometry_msgs tf2 tf2_ros stretch_robot_interface)
ament_package()
